local function startFly()
    if not MyPlayer.Character or not MyPlayer.Character:FindFirstChild("Head") or flyState.flying then return end
    flyState.c = MyPlayer.Character
    flyState.h = flyState.c:FindFirstChildOfClass("Humanoid")
    if not flyState.h then return end
    pcall(function() flyState.h.PlatformStand = true end)
    flyState.cam = WorkspaceService.CurrentCamera
    flyState.bv = Instance.new("BodyVelocity")
    flyState.bav = Instance.new("BodyAngularVelocity")
    flyState.bv.Velocity = Vector3.new(0, 0, 0)
    flyState.bv.MaxForce = Vector3.new(10000, 10000, 10000)
    flyState.bv.P = 1000
    flyState.bav.AngularVelocity = Vector3.new(0, 0, 0)
    flyState.bav.MaxTorque = Vector3.new(10000, 10000, 10000)
    flyState.bav.P = 1000
    flyState.bv.Parent = flyState.c.Head
    flyState.bav.Parent = flyState.c.Head
    flyState.flying = true
    flyState.h.Died:Connect(function() flyState.flying = false end)
end

local function endFly()
    if flyState.h then pcall(function() flyState.h.PlatformStand = false end) end
    if flyState.bv then pcall(function() flyState.bv:Destroy() end) flyState.bv = nil end
    if flyState.bav then pcall(function() flyState.bav:Destroy() end) flyState.bav = nil end
    flyState.flying = false
    flyState.c = nil
    flyState.h = nil
    for k in pairs(flyState.buttons) do flyState.buttons[k] = false end
    for _, btn in pairs(mobileButtons) do
        pcall(function() btn.BackgroundColor3 = Color3.fromRGB(30, 30, 30) end)
    end
end

local function setVec(vec)
    if vec.Magnitude == 0 then return Vector3.new() end
    return vec * (flySettings.flyspeed / vec.Magnitude)
end

RunServiceRef.Heartbeat:Connect(function(step)
    if flyState.flying and flyState.c and flyState.c.PrimaryPart then
        local p = flyState.c.PrimaryPart.Position
        local cf = flyState.cam.CFrame
        local ax, ay, az = cf:ToEulerAnglesXYZ()
        flyState.c:PivotTo(CFrame.new(p.X, p.Y, p.Z) * CFrame.Angles(ax, ay, az))
        if flyState.buttons.Moving then
            local t = Vector3.new()
            if flyState.buttons.W then t = t + setVec(cf.LookVector) end
            if flyState.buttons.S then t = t - setVec(cf.LookVector) end
            if flyState.buttons.A then t = t - setVec(cf.RightVector) end
            if flyState.buttons.D then t = t + setVec(cf.RightVector) end
            flyState.c:TranslateBy(t * step)
        end
    end
end)

local walkState = { enabled = false, speed = 16, method = "Velocity" }

RunServiceRef.Stepped:Connect(function(dt)
    if not walkState.enabled then return end
    if flyState.flying then return end
    local char = MyPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    local VS = hum.MoveDirection * walkState.speed
    if walkState.method == "Velocity" then
        root.Velocity = Vector3.new(VS.X, root.Velocity.Y, VS.Z)
    elseif walkState.method == "Vector" then
        root.CFrame = root.CFrame + (VS * dt)
    elseif walkState.method == "CFrame" then
        root.CFrame = root.CFrame + (hum.MoveDirection * walkState.speed * dt)
    else
        hum.WalkSpeed = walkState.speed
    end
end)

local jumpState = { enabled = false, power = 30, method = "Velocity" }
local jumpConnections = {}

local function hookJump(char)
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local conn
    conn = hum.Jumping:Connect(function(isActive)
        if not jumpState.enabled or not isActive then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        if jumpState.method == "Velocity" then
            root.Velocity = root.Velocity * Vector3.new(1, 0, 1) + Vector3.new(0, jumpState.power, 0)
        elseif jumpState.method == "Vector" then
            root.Velocity = Vector3.new(0, jumpState.power, 0)
        elseif jumpState.method == "CFrame" then
            if char.PrimaryPart then
                char:PivotTo(char:GetPivot() + Vector3.new(0, jumpState.power, 0))
            end
        end
    end)
    table.insert(jumpConnections, conn)
end

MyPlayer.CharacterAdded:Connect(hookJump)

local function triggerIsEnemy(targetPlayer)
    if triggerState.teamCheck == "FFA" then return true
    elseif triggerState.teamCheck == "Everyone" then return targetPlayer ~= MyPlayer
    elseif triggerState.teamCheck == "Team-Based" then return targetPlayer.Team ~= MyPlayer.Team end
    return false
end

local function hasLOS(origin, target)
    local dir = target - origin
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = { MyPlayer.Character }
    rp.IgnoreWater = true
    local res = WorkspaceService:Raycast(origin, dir, rp)
    return res == nil
end

    if hum then
        triggerState.alive = hum.Health > 0
        hum.HealthChanged:Connect(function(h) triggerState.alive = h > 0 end)
    end
end

local function getBestTargetInFov()
    local cam = WorkspaceService.CurrentCamera
    if not cam then return nil end
    local myChar = MyPlayer.Character
    if not myChar then return nil end
    local myHead = myChar:FindFirstChild("Head")
    if not myHead then return nil end
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
    local best, bestDist = nil, triggerState.fov + 1
    for _, plr in ipairs(PlayersService:GetPlayers()) do
        if plr ~= MyPlayer and triggerIsEnemy(plr) then
            local char = plr.Character
            if char then
                local hum = char:FindFirstChildOfClass("Humanoid")
                local head = char:FindFirstChild("Head")
                if hum and hum.Health > 0 and head then
                    if triggerState.wallCheck and not hasLOS(myHead.Position, head.Position) then
                        continue
                    end
                    local screenPos, onScreen = cam:WorldToScreenPoint(head.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                        if dist < bestDist then
                            bestDist = dist
                            best = plr
                        end
                    end
                end
            end
        end
    end
    return best
end

local hitboxState = {
    enabled = false, size = 21, transparency = 6,
    teamCheck = "FFA", noCollision = false, original = {}
}
local hitboxParts = { "HeadHB", "LowerTorso", "HumanoidRootPart" }

local function savedPart(player, part)
    if not hitboxState.original[player] then hitboxState.original[player] = {} end
    if not hitboxState.original[player][part.Name] then
        hitboxState.original[player][part.Name] = {
            CanCollide = part.CanCollide,
            Transparency = part.Transparency,
            Size = part.Size
        }
    end
end

local function restoredPart(player)
    if hitboxState.original[player] then
        for partName, props in pairs(hitboxState.original[player]) do
            local part = player.Character and player.Character:FindFirstChild(partName)
            if part and part:IsA("BasePart") then
                part.CanCollide = props.CanCollide
                part.Transparency = props.Transparency
                part.Size = props.Size
            end
        end
    end
end

local function isEnemyHb(player)
    if hitboxState.teamCheck == "FFA" or hitboxState.teamCheck == "Everyone" then return true end
    return player.Team ~= MyPlayer.Team
end

local function extendHitbox(player)
    for _, partName in ipairs(hitboxParts) do
        local part = player.Character and player.Character:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            savedPart(player, part)
            part.CanCollide = not hitboxState.noCollision
            part.Transparency = hitboxState.transparency / 10
            part.Size = Vector3.new(hitboxState.size, hitboxState.size, hitboxState.size)
        end
    end
end

local function updateHitboxes()
    for _, v in ipairs(PlayersService:GetPlayers()) do
        if v ~= MyPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
            if isEnemyHb(v) then extendHitbox(v) else restoredPart(v) end
        end
    end
end

task.spawn(function()
    while task.wait(0.1) do
        if hitboxState.enabled then updateHitboxes() end
    end
end)

local originalValues = { FireRate = {}, ReloadTime = {}, EReloadTime = {}, Auto = {}, Spread = {}, Recoil = {} }

local function enableFastReload(state)
    local weapons = ReplicatedStorage:FindFirstChild("Weapons")
    if not weapons then return end
    for _, v in ipairs(weapons:GetChildren()) do
        if v:FindFirstChild("ReloadTime") then
            if state then
                if not originalValues.ReloadTime[v] then originalValues.ReloadTime[v] = v.ReloadTime.Value end
                v.ReloadTime.Value = 0.01
            else
                v.ReloadTime.Value = originalValues.ReloadTime[v] or 0.8
            end
        end
        if v:FindFirstChild("EReloadTime") then
            if state then
                if not originalValues.EReloadTime[v] then originalValues.EReloadTime[v] = v.EReloadTime.Value end
                v.EReloadTime.Value = 0.01
            else
                v.EReloadTime.Value = originalValues.EReloadTime[v] or 0.8
            end
        end
    end
end

local function enableFastFire(state)

    if not weapons then return end
    for _, v in ipairs(weapons:GetDescendants()) do
        if v.Name == "FireRate" or v.Name == "BFireRate" then

                v.Value = 0.02

local function createEspFolder()
    if espState.folder and espState.folder.Parent then return espState.folder end
    local folder = Instance.new("Folder")
    folder.Name = "AureusESP"
    pcall(function() folder.Parent = CoreGuiService end)
    if not folder.Parent then folder.Parent = MyPlayer:WaitForChild("PlayerGui") end
    espState.folder = folder
    return folder
end

local function clearEspFolder()
    if espState.folder then
        for _, child in ipairs(espState.folder:GetChildren()) do
            pcall(function() child:Destroy() end)
        end
    end
end

local function isEspEnemy(player)
    if not espState.teamCheck then return player ~= MyPlayer end
    return player.Team ~= MyPlayer.Team
end

local function updateEsp()
    if not espState.enabled then return end
    local folder = createEspFolder()
    for _, plr in ipairs(PlayersService:GetPlayers()) do
        if plr ~= MyPlayer and isEspEnemy(plr) and plr.Character then
            local char = plr.Character
            local hum = char:FindFirstChildOfClass("Humanoid")
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and hrp then
                local existing = folder:FindFirstChild(plr.Name)
                if not existing then
                    existing = Instance.new("Folder")
                    existing.Name = plr.Name
                    existing.Parent = folder
                end

                if espState.method == "Highlight" or espState.method == "Box" then
                    local highlight = existing:FindFirstChild("Highlight")
                    if not highlight then
                        highlight = Instance.new("Highlight")
                        highlight.Name = "Highlight"
                        highlight.FillTransparency = 0.6
                        highlight.OutlineTransparency = 0.3
                        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        highlight.Parent = existing
                    end
                    highlight.Adornee = char
                    highlight.FillColor = espState.color
                    highlight.OutlineColor = espState.color
                else
                    local h = existing:FindFirstChild("Highlight")
                    if h then h:Destroy() end
                end

                if espState.showNames or espState.showHealth or espState.showDistance then
                    local billboard = existing:FindFirstChild("Info")
                    if not billboard then
                        billboard = Instance.new("BillboardGui")
                        billboard.Name = "Info"
                        billboard.Size = UDim2.fromOffset(200, 50)
                        billboard.StudsOffset = Vector3.new(0, 3.5, 0)
                        billboard.AlwaysOnTop = true
                        billboard.Parent = existing
                    end
                    billboard.Adornee = char:FindFirstChild("Head")

                    local nameLabel = billboard:FindFirstChild("NameLabel")
                    if espState.showNames then
                        if not nameLabel then
                            nameLabel = Instance.new("TextLabel")
                            nameLabel.Name = "NameLabel"
                            nameLabel.BackgroundTransparency = 1
                            nameLabel.Size = UDim2.new(1, 0, 0.4, 0)
                            nameLabel.Font = Enum.Font.GothamBold
                            nameLabel.TextScaled = true
                            nameLabel.TextStrokeTransparency = 0.3
                            nameLabel.TextColor3 = Color3.new(1, 1, 1)
                            nameLabel.Parent = billboard
                        end
                        nameLabel.Text = plr.Name
                        nameLabel.TextColor3 = espState.color
                    elseif nameLabel then
                        nameLabel:Destroy()
                    end

                    local healthBar = billboard:FindFirstChild("HealthBar")
                    if espState.showHealth then
                        if not healthBar then
                            healthBar = Instance.new("Frame")
                            healthBar.Name = "HealthBar"
                            healthBar.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
                            healthBar.BorderSizePixel = 0
                            healthBar.Size = UDim2.new(1, 0, 0.25, 0)
                            healthBar.Position = UDim2.new(0, 0, 0.45, 0)
                            healthBar.Parent = billboard
                            local bg = Instance.new("Frame")
                            bg.Name = "BG"
                            bg.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
                            bg.BorderSizePixel = 0
                            bg.Size = UDim2.new(1, 0, 1, 0)
                            bg.ZIndex = 0
                            bg.Parent = healthBar
                        end
                        local hpRatio = hum.Health / hum.MaxHealth
                        healthBar.Size = UDim2.new(hpRatio, 0, 0.25, 0)
                        if hpRatio > 0.5 then
                            healthBar.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
                        elseif hpRatio > 0.25 then
                            healthBar.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
                        else
                            healthBar.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
                        end
                    elseif healthBar then
                        healthBar:Destroy()
                    end

                    local distLabel = billboard:FindFirstChild("DistLabel")
                    if espState.showDistance then
                        if not distLabel then
                            distLabel = Instance.new("TextLabel")
                            distLabel.Name = "DistLabel"
                            distLabel.BackgroundTransparency = 1
                            distLabel.Size = UDim2.new(1, 0, 0.3, 0)
                            distLabel.Position = UDim2.new(0, 0, 0.7, 0)
                            distLabel.Font = Enum.Font.Gotham
                            distLabel.TextScaled = true
                            distLabel.TextColor3 = Color3.new(1, 1, 1)
                            distLabel.Parent = billboard
                        end
                        local myHrp = MyPlayer.Character and MyPlayer.Character:FindFirstChild("HumanoidRootPart")
                        if myHrp then
                            distLabel.Text = math.floor((myHrp.Position - hrp.Position).Magnitude) .. "m"
                        end
                    elseif distLabel then
                        distLabel:Destroy()
                    end
                else
                    local b = existing:FindFirstChild("Info")
                    if b then b:Destroy() end
                end

                if espState.showTracers then

                if e then e:Destroy() end
            end
        else
            local e = folder:FindFirstChild(plr.Name)

local antiLagActive = false
local antiLagConnection = nil
local antiLagProcessed = setmetatable({}, { __mode = "k" })

local function isMine(obj)
    if not obj then return false end
    local myChar = MyPlayer.Character
    if myChar and obj:IsDescendantOf(myChar) then return true end
    return false
end

local function antiLagProcessOne(obj)
    if not antiLagActive or not obj or not obj.Parent then return end
    if antiLagProcessed[obj] then return end
    antiLagProcessed[obj] = true
    if isMine(obj) then return end
    if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam")
        or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
        pcall(function() obj.Enabled = false end)
        return
    end
    if obj:IsA("BasePart") then
        local model = obj:FindFirstAncestorOfClass("Model")
        if not (model and model:FindFirstChildOfClass("Humanoid")) then
            pcall(function()
                if obj.CastShadow then obj.CastShadow = false end
                obj.Material = Enum.Material.SmoothPlastic
                if obj.Reflectance > 0 then obj.Reflectance = 0 end
            end)
        end
    end
end

local function antiLagOn()
    if antiLagActive then return end
    antiLagActive = true
    if antiLagConnection then pcall(function() antiLagConnection:Disconnect() end) end
    antiLagConnection = WorkspaceService.DescendantAdded:Connect(function(obj)
        if not antiLagActive then return end
        task.defer(function() antiLagProcessOne(obj) end)
    end)
    task.spawn(function()
        local n = 0
        for _, obj in ipairs(WorkspaceService:GetDescendants()) do
            if not antiLagActive then return end
            antiLagProcessOne(obj)
            n = n + 1
            if n % 500 == 0 then task.wait() end
        end
    end)
end

local function antiLagOff()
    antiLagActive = false
    if antiLagConnection then pcall(function() antiLagConnection:Disconnect() end) antiLagConnection = nil end
    antiLagProcessed = setmetatable({}, { __mode = "k" })
end

pcall(function() getgenv().AureusAntiLagOff = antiLagOff end)

local function fpsBoostOn()
    local t = WorkspaceService.Terrain
    t.WaterWaveSize = 0
    t.WaterWaveSpeed = 0
    t.WaterReflectance = 0
    t.WaterTransparency = 0
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    Lighting.Brightness = 0
    pcall(function() settings().Rendering.QualityLevel = "Level01" end)
    for _, v in ipairs(game:GetDescendants()) do
        if v:IsA("Part") or v:IsA("Union") or v:IsA("CornerWedgePart") or v:IsA("TrussPart") then
            fpsState.origMaterials[v] = v.Material
            v.Material = Enum.Material.Plastic
            v.Reflectance = 0
        elseif v:IsA("Decal") or v:IsA("Texture") then
            table.insert(fpsState.origDecals, v)
            v.Transparency = 1
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") then
            v.Lifetime = NumberRange.new(0)
        elseif v:IsA("Explosion") then
            v.BlastPressure = 1
            v.BlastRadius = 1
        elseif v:IsA("Fire") or v:IsA("SpotLight") or v:IsA("Smoke") then
            v.Enabled = false
        elseif v:IsA("MeshPart") then
            fpsState.origMaterials[v] = v.Material
            v.Material = Enum.Material.Plastic
            v.Reflectance = 0
        end
    end
    for _, e in ipairs(Lighting:GetChildren()) do
        if e:IsA("BlurEffect") or e:IsA("SunRaysEffect") or e:IsA("ColorCorrectionEffect") or e:IsA("BloomEffect") or e:IsA("DepthOfFieldEffect") then
            fpsState.origEffects[e] = e.Enabled
            e.Enabled = false
        end
    end
end

local function fpsBoostOff()
    local t = WorkspaceService.Terrain
    t.WaterWaveSize = fpsState.origTerrain.WaterWaveSize
    t.WaterWaveSpeed = fpsState.origTerrain.WaterWaveSpeed
    t.WaterReflectance = fpsState.origTerrain.WaterReflectance
    t.WaterTransparency = fpsState.origTerrain.WaterTransparency
    Lighting.GlobalShadows = fpsState.origLighting.GlobalShadows
    Lighting.FogEnd = fpsState.origLighting.FogEnd
    Lighting.Brightness = fpsState.origLighting.Brightness
    pcall(function() settings().Rendering.QualityLevel = "Automatic" end)
    for v, mat in pairs(fpsState.origMaterials) do
        if v and v.Parent and v:IsA("BasePart") then
            v.Material = mat
            v.Reflectance = 0
        end
    end
    fpsState.origMaterials = {}
    for e, en in pairs(fpsState.origEffects) do
        if e and e.Parent then e.Enabled = en end
    end
    fpsState.origEffects = {}
    for _, v in ipairs(fpsState.origDecals) do
        if v and v.Parent then v.Transparency = 0 end
    end
    fpsState.origDecals = {}
end

local function serverHop()
    local placeID = game.PlaceId
    local allIDs = {}
    local foundAnything = ""
    local actualHour = os.date("!*t").hour
    local file = pcall(function() allIDs = game:GetService("HttpService"):JSONDecode(readfile("AureusNotSameServers.json")) end)
    if not file then
        table.insert(allIDs, actualHour)
        writefile("AureusNotSameServers.json", game:GetService("HttpService"):JSONEncode(allIDs))
    end
    local function teleportReturner()
        local site
        if foundAnything == "" then
            site = game:GetService("HttpService"):JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. placeID .. "/servers/Public?sortOrder=Asc&limit=100"))
        else
            site = game:GetService("HttpService"):JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. placeID .. "/servers/Public?sortOrder=Asc&limit=100&cursor=" .. foundAnything))
        end
        local serverID = ""
        if site.nextPageCursor and site.nextPageCursor ~= "null" and site.nextPageCursor ~= nil then
            foundAnything = site.nextPageCursor
        end
        local num = 0
        for _, v in pairs(site.data) do
            local possible = true
            serverID = tostring(v.id)
            if tonumber(v.maxPlayers) > tonumber(v.playing) then
                for _, existing in pairs(allIDs) do
                    if num ~= 0 then
                        if serverID == tostring(existing) then possible = false end
                    else
                        if tonumber(actualHour) ~= tonumber(existing) then
                            pcall(function()
                                delfile("AureusNotSameServers.json")
                                allIDs = {}
                                table.insert(allIDs, actualHour)
                            end)
                        end
                    end
                    num = num + 1
                end
                if possible then
                    table.insert(allIDs, serverID)
                    task.wait()
                    pcall(function()
                        writefile("AureusNotSameServers.json", game:GetService("HttpService"):JSONEncode(allIDs))
                        task.wait()
                        game:GetService("TeleportService"):TeleportToPlaceInstance(placeID, serverID, MyPlayer)
                    end)
                    task.wait(4)
                end
            end
        end
    end
    task.spawn(function()
        while true do
            pcall(function()
                teleportReturner()
                if foundAnything ~= "" then teleportReturner() end
            end)
            task.wait()
        end
    end)
end

    CurrentValue = false,
    Flag = "NameChange",
    Callback = function(v)
        changeNameRunning = v
        if v then
            pcall(function()
                local gui = MyPlayer.PlayerGui
                if gui:FindFirstChild("Menew_Main") then
                    originalNames.GUIName = gui.Menew_Main.Container.PlrName.Text
                    originalNames.GUIName2 = gui.Menew_Main.Container.PlrName2.Text
                end
                if gui:FindFirstChild("GUI") and gui.GUI:FindFirstChild("Winner") then
                    originalNames.WinnerName = gui.GUI.Winner.Visible
                end
                if gui:FindFirstChild("GUI_Scorecard") then
                    originalNames.ScorecardName = gui.GUI_Scorecard.Scorecard.PlayerCard.Username.Text
                end
                local kf = WorkspaceService:FindFirstChild("KillFeed")
                if kf then
                    for i = 1, 6 do
                        local slot = kf[tostring(i)]
                        if slot and slot:FindFirstChild("Killer") then originalNames.KillFeed[i] = slot.Killer.Value end
                    end
                end
            end)
            task.spawn(function()
                while changeNameRunning do
                    applyNameChange()
                    task.wait(0.2)
                end
            end)
        else
            restoreNameChange()
        end
    end,
})
